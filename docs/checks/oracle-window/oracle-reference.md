## `OracleAttestation.sol`

The library and the base contract. Copy it whole into your `src/`; inherit `OracleAttestationConsumer`. It needs OpenZeppelin's `EIP712` and `SignatureChecker`.

```solidity
// SPDX-License-Identifier: MIT
pragma solidity 0.8.26;

import {EIP712} from "@openzeppelin/contracts/utils/cryptography/EIP712.sol";
import {SignatureChecker} from "@openzeppelin/contracts/utils/cryptography/SignatureChecker.sol";

/// @title The reasoning oracle's attestation, as a contract reads it
/// @notice The oracle answers a typed question off chain — a panel of agents reads it, a deployer
/// reproduces the answer from the chain — and signs the result as EIP-712 typed data in the
/// *consumer's* domain: this chain, this contract. A signature for one consumer is meaningless to
/// another, which is the property that lets one attester key serve every consumer without any of
/// them being able to replay an answer into a neighbour.
///
/// @dev This library is the Solidity half of `oracle-eip712.ts` in `@identitymd/protocol`. The
/// struct, the type string and the domain name are copied from there field for field, and the
/// cross-implementation vector in `test/OracleAttestation.t.sol` is what keeps them equal: a digest
/// computed by viem on one side and `_hashTypedDataV4` on the other, over the same values. Change
/// either half and the vector fails before anything is signed for real.
library OracleAttestation {
    /// @dev Field order is the typed data's. `answer` is `abi.encode` of the value under
    /// `answerType`, so a consumer reads it back with one `abi.decode`; the hash covers
    /// `keccak256(answer)` as EIP-712 requires for a dynamic field.
    struct Attestation {
        /// @dev The request's UUID as sixteen raw bytes, left-aligned. The natural replay key.
        bytes32 requestId;
        /// @dev The chain the question is *about* — the one the recipe ran on. Not necessarily the
        /// consumer's chain, which is in the domain instead.
        uint256 chainId;
        /// @dev keccak-256 of the canonical question document. A consumer that pins its question
        /// compares this, so an answer to a different question cannot be presented as its own.
        bytes32 questionHash;
        uint8 answerType;
        bytes answer;
        /// @dev The figure behind the answer where there is one: the sum, the leader's volume, the
        /// value compared. Zero otherwise.
        uint256 figure;
        uint64 fromBlock;
        uint64 toBlock;
        /// @dev The closing block's hash, so a reorg cannot quietly change what was answered.
        bytes32 blockHash;
        /// @dev The panel job's UUID, the same way. Its page and receipt hold the evidence.
        bytes32 panelJobId;
        /// @dev How many seats the request opened. With `quorum`, what the requester asked for, so a
        /// consumer that does not pin its question can still refuse a panel of two.
        uint16 panelSize;
        /// @dev How many answers had to agree for the panel to settle.
        uint16 quorum;
        /// @dev How many members gave the answer signed. At least `quorum` when the panel agreed;
        /// below it only for chain evidence, when members who ran one recipe split and the
        /// deployer's own rerun settled which of their answers was whole. A consumer that wants the
        /// panel's agreement itself, not the rerun's, requires `agreed >= quorum`.
        uint16 agreed;
        uint64 issuedAt;
        uint64 expiresAt;
    }

    /// @dev The `answerType` codes. They are the protocol enum's order and are appended to, never
    /// reordered, because a code is what a signed attestation carries.
    uint8 internal constant ANSWER_BOOL = 0;
    uint8 internal constant ANSWER_ADDRESS = 1;
    uint8 internal constant ANSWER_BYTES32 = 2;
    uint8 internal constant ANSWER_UINT256 = 3;
    uint8 internal constant ANSWER_ADDRESS_LIST = 4;
    uint8 internal constant ANSWER_BYTES32_LIST = 5;

    string internal constant DOMAIN_NAME = "IdentityMD Oracle";
    /// @dev Version 2 added `panelSize`, `quorum` and `agreed`. A version 1 signature does not
    /// verify here and a version 2 one does not verify against a version 1 consumer.
    string internal constant DOMAIN_VERSION = "2";

    bytes32 internal constant TYPEHASH = keccak256(
        "OracleAttestation(bytes32 requestId,uint256 chainId,bytes32 questionHash,uint8 answerType,bytes answer,uint256 figure,uint64 fromBlock,uint64 toBlock,bytes32 blockHash,bytes32 panelJobId,uint16 panelSize,uint16 quorum,uint16 agreed,uint64 issuedAt,uint64 expiresAt)"
    );

    /// @notice The EIP-712 `hashStruct` of an attestation: what goes under the domain separator.
    /// @dev Encoded in two halves because sixteen values in one `abi.encode` is too deep a stack
    /// for the legacy code generator, and a consumer should not need via-IR to inherit this. Every
    /// value is a static 32-byte word, so the two halves concatenated are byte for byte the one
    /// encoding EIP-712 specifies.
    function hashStruct(Attestation calldata a) internal pure returns (bytes32) {
        return keccak256(
            bytes.concat(
                abi.encode(
                    TYPEHASH,
                    a.requestId,
                    a.chainId,
                    a.questionHash,
                    a.answerType,
                    keccak256(a.answer),
                    a.figure,
                    a.fromBlock
                ),
                abi.encode(
                    a.toBlock, a.blockHash, a.panelJobId, a.panelSize, a.quorum, a.agreed, a.issuedAt, a.expiresAt
                )
            )
        );
    }
}

/// @title What a contract inherits to accept oracle attestations
/// @notice Verifies that an attestation was signed by this consumer's oracle signer, for this
/// consumer, and is inside its validity window. What the consumer then *does* with the answer is
/// its own business: this contract decodes, it never acts.
///
/// @dev Replay is deliberately not tracked here. Some consumers want one action per request; some
/// want to read the same attestation from several functions; some key on `questionHash` rather than
/// `requestId`. `consumed` and `_consume` are offered for the first kind and cost nothing to the
/// others. A consumer that neither consumes nor keys on something else **accepts the same signed
/// answer as many times as it is presented** — for as long as it is valid — and must want that.
///
/// The signer is one address, so the natural rotation story is to point it at an
/// `OracleSignerRegistry`, which answers ERC-1271 for the keys it lists; `SignatureChecker` takes
/// that path for any signer with code. A consumer that prefers to hold a bare key overrides
/// nothing and rotates through `_setOracleSigner`.
abstract contract OracleAttestationConsumer is EIP712 {
    using OracleAttestation for OracleAttestation.Attestation;

    error AttestationExpired(uint64 expiresAt);
    error AttestationNotYetValid(uint64 issuedAt);
    error BadSignature();
    error AlreadyConsumed(bytes32 requestId);
    error WrongAnswerType(uint8 expected, uint8 got);
    error ZeroSigner();

    event OracleSignerSet(address indexed signer);

    /// @dev How far ahead of this chain's clock an `issuedAt` may sit. The attester stamps with
    /// its own wall clock; a block's timestamp is a validator's. Five minutes is generous for
    /// honest drift and still short of anything that could be called pre-signing.
    uint64 public constant ISSUED_AT_TOLERANCE = 5 minutes;

    /// @notice The address every attestation must verify against: a key, or a registry of keys.
    address public oracleSigner;

    /// @notice Request ids this consumer has acted on, for consumers that call `_consume`.
    mapping(bytes32 => bool) public consumed;

    constructor(address signer) EIP712(OracleAttestation.DOMAIN_NAME, OracleAttestation.DOMAIN_VERSION) {
        _setOracleSigner(signer);
    }

    /// @notice The digest the oracle signed: this consumer's domain over the attestation.
    /// @dev Public so an off-chain party — the attester, a test, a reader — can ask the consumer
    /// what it will check rather than reimplementing the domain. Equal to `oracleAttestationDigest`
    /// in the protocol package for the same consumer and message.
    function attestationDigest(OracleAttestation.Attestation calldata a) public view returns (bytes32) {
        return _hashTypedDataV4(a.hashStruct());
    }

    /// @notice Reverts unless the attestation is current and signed for this consumer.
    /// @dev Checks the window first because a stale signature is the common failure and the cheap
    /// one to report. Does not check `a.chainId` — the chain the question was about is a fact
    /// about the question, and whether it is the right one is for the consumer to decide alongside
    /// `questionHash`. A validator can nudge `block.timestamp` by seconds; the window is minutes
    /// to days wide and the tolerance absorbs the other end, so that is not a lever here.
    function _verifyAttestation(OracleAttestation.Attestation calldata a, bytes calldata signature) internal view {
        // forge-lint: disable-next-line(block-timestamp)
        if (block.timestamp > a.expiresAt) revert AttestationExpired(a.expiresAt);
        // forge-lint: disable-next-line(block-timestamp)
        if (a.issuedAt > block.timestamp + ISSUED_AT_TOLERANCE) revert AttestationNotYetValid(a.issuedAt);
        if (!SignatureChecker.isValidSignatureNowCalldata(oracleSigner, attestationDigest(a), signature)) {
            revert BadSignature();
        }
    }

    /// @notice Marks a request id as acted on, once.
    /// @dev Opt-in. Call it after `_verifyAttestation` and before the effect, in the same function,
    /// so a re-presented attestation fails here rather than repeating the effect.
    function _consume(bytes32 requestId) internal {
        if (consumed[requestId]) revert AlreadyConsumed(requestId);
        consumed[requestId] = true;
    }

    /// @dev The hook a consumer wires to whatever authority it has. Not exposed here: this contract
    /// does not know who is allowed to rotate, and an unguarded public setter would be the whole
    /// oracle's key in anyone's hands. The zero address is refused because a consumer with no
    /// signer only finds out on its first attestation, with `BadSignature`, which points nowhere.
    function _setOracleSigner(address signer) internal {
        if (signer == address(0)) revert ZeroSigner();
        oracleSigner = signer;
        emit OracleSignerSet(signer);
    }

    function decodeBool(OracleAttestation.Attestation calldata a) internal pure returns (bool) {
        _expectType(a, OracleAttestation.ANSWER_BOOL);
        return abi.decode(a.answer, (bool));
    }

    function decodeAddress(OracleAttestation.Attestation calldata a) internal pure returns (address) {
        _expectType(a, OracleAttestation.ANSWER_ADDRESS);
        return abi.decode(a.answer, (address));
    }

    function decodeBytes32(OracleAttestation.Attestation calldata a) internal pure returns (bytes32) {
        _expectType(a, OracleAttestation.ANSWER_BYTES32);
        return abi.decode(a.answer, (bytes32));
    }

    function decodeUint256(OracleAttestation.Attestation calldata a) internal pure returns (uint256) {
        _expectType(a, OracleAttestation.ANSWER_UINT256);
        return abi.decode(a.answer, (uint256));
    }

    function decodeAddressList(OracleAttestation.Attestation calldata a) internal pure returns (address[] memory) {
        _expectType(a, OracleAttestation.ANSWER_ADDRESS_LIST);
        return abi.decode(a.answer, (address[]));
    }

    function decodeBytes32List(OracleAttestation.Attestation calldata a) internal pure returns (bytes32[] memory) {
        _expectType(a, OracleAttestation.ANSWER_BYTES32_LIST);
        return abi.decode(a.answer, (bytes32[]));
    }

    /// @dev The type is signed, so decoding under another is not a parsing accident to tolerate:
    /// an address read as a uint256 is a number that means nothing and a uint256 read as an address
    /// is twelve bytes of somebody's balance, silently dropped.
    function _expectType(OracleAttestation.Attestation calldata a, uint8 expected) private pure {
        if (a.answerType != expected) revert WrongAnswerType(expected, a.answerType);
    }
}
```

## `OracleQuestion.sol`

Computes a question's `questionHash` on chain, and the request body from the same values, for a contract that pins the questions it asks. Copy it whole beside `OracleAttestation.sol`; it needs OpenZeppelin's `Strings`.

```solidity
// SPDX-License-Identifier: MIT
pragma solidity 0.8.26;

import {Strings} from "@openzeppelin/contracts/utils/Strings.sol";
import {OracleAttestation} from "./OracleAttestation.sol";

/// @title The oracle question's hash, computed on chain
/// @notice Every attestation signs a `questionHash`: keccak-256 of the canonical question document,
/// which holds the question, its chain, its block window, its answer type and its definitions. A
/// contract that computes that hash itself, before it asks, can refuse every answer to any other
/// question. Store the hash beside the pending request and compare `a.questionHash` in the
/// callback: the signature then says which request the answer is for, and an answer paired with
/// the wrong request, by mistake or on purpose, is refused rather than acted on.
///
/// The hash is known in advance only for an absolute window (`{"fromBlock", "toBlock"}`) on the
/// question's chain, whose last block is at least five blocks deep when the plane reads the ask.
/// A relative window (`{"hours": N}`) resolves to blocks only then. On an Arbitrum chain, which
/// Robinhood Chain is, `block.number` is an Ethereum block number: read the chain's own from
/// `ArbSys(address(100)).arbBlockNumber()`.
///
/// This computes the hash; it does not judge whether the plane accepts the question. Check a body
/// with a real subject in it at `POST /requests/check` before a contract is deployed with it.
///
/// @dev The Solidity half of `questionDocument` and `canonicalize` in `@identitymd/protocol`:
/// object keys sorted, strings as `JSON.stringify` writes them, integers in decimal. The same
/// documents and hashes are pinned in `test/OracleQuestion.t.sol` and in the protocol package's
/// `test/oracle.test.ts`, so a change to either half fails a test before anything is signed.
///
/// It refuses what it cannot reproduce exactly, rather than hash bytes the plane will never hash:
/// - text outside ASCII. The plane reads a body's bytes as UTF-8, replacing what is not, and only
///   ASCII is checked here;
/// - a value that is exactly an address or a 32-byte hex string with an uppercase digit. The intake
///   lowercases such a value before anything reads it; hex inside longer text is left as written;
/// - definitions out of order. The document sorts its keys, so they are taken already sorted,
///   byte by byte, which is JavaScript's order for ASCII keys; a repeated key is out of order too;
/// - a number above 2^53 - 1, which JSON cannot carry exactly to the plane.
library OracleQuestion {
    /// @dev One of the definitions the requester pins: what a word in the question means.
    struct Definition {
        string key;
        string value;
    }

    struct Question {
        string question;
        /// @dev The chain the question is about, and whose blocks the window counts.
        uint256 chainId;
        uint256 fromBlock;
        uint256 toBlock;
        /// @dev One of `OracleAttestation`'s `ANSWER_*` codes.
        uint8 answerType;
        /// @dev For a list answer: how many leading positions are used. Zero for none.
        uint8 head;
        /// @dev `"evidence": "panel"`, a fact that is not on any chain. False for chain evidence,
        /// the default, which the document leaves out.
        bool panelEvidence;
        /// @dev Ascending by key. Empty for none: then the body must leave `definitions` out too,
        /// since `"definitions": {}` is a member of the document.
        Definition[] definitions;
    }

    error UnsupportedCharacter(uint256 index);
    error NotLowercaseHex();
    error DefinitionsOutOfOrder(uint256 index);
    error UnknownAnswerType(uint8 answerType);
    error NumberTooLarge(uint256 value);
    error WindowReversed(uint256 fromBlock, uint256 toBlock);

    /// @dev The largest integer a JSON number carries exactly to the plane: `Number.MAX_SAFE_INTEGER`.
    uint256 internal constant MAX_SAFE_INTEGER = 9_007_199_254_740_991;

    bytes16 private constant HEX_DIGITS = "0123456789abcdef";

    /// @notice The `questionHash` an attestation for this question carries.
    function hash(Question memory q) internal pure returns (bytes32) {
        return keccak256(document(q));
    }

    /// @notice The canonical question document, the bytes `hash` covers.
    function document(Question memory q) internal pure returns (bytes memory) {
        return bytes.concat(_members(q), "}");
    }

    /// @notice A complete `oracle.request` body for this question: the document's members, then the
    /// panel and how long the answer stays valid. The body and the hash come from the same values,
    /// so the question the plane hashes is the question this contract pinned. A body that needs
    /// more members (`allowAmbiguous`, `toleranceBps`, `consumer`) is built the same way, with
    /// `jsonString` for any text.
    function body(Question memory q, uint256 panelSize, uint256 quorum, uint256 validForSeconds)
        internal
        pure
        returns (bytes memory)
    {
        return bytes.concat(
            _members(q),
            ',"panelSize":',
            _number(panelSize),
            ',"quorum":',
            _number(quorum),
            ',"validForSeconds":',
            _number(validForSeconds),
            "}"
        );
    }

    /// @notice `text` as `JSON.stringify` writes it, quotes included: a quote, a backslash and the
    /// control characters escaped, everything else as it is. Reverts on a byte outside ASCII.
    function jsonString(string memory text) internal pure returns (bytes memory quoted) {
        bytes memory raw = bytes(text);
        uint256 length = raw.length;
        // How many bytes the escapes add. Most text has none, and is copied whole below.
        uint256 added = 0;
        unchecked {
            for (uint256 i; i < length; ++i) {
                uint256 c = _byteAt(raw, i);
                if (c >= 0x80) revert UnsupportedCharacter(i);
                if (c < 0x20) added += _shortEscape(c) != 0 ? 1 : 5;
                else if (c == 0x22 || c == 0x5c) added += 1;
            }
        }
        if (added == 0) return bytes.concat('"', raw, '"');

        // Sized exactly by the count above, so every write lands inside it.
        quoted = new bytes(length + added + 2);
        unchecked {
            uint256 j = _put(quoted, 0, 0x22);
            for (uint256 i; i < length; ++i) {
                uint256 c = _byteAt(raw, i);
                if (c >= 0x20 && c != 0x22 && c != 0x5c) {
                    j = _put(quoted, j, c);
                    continue;
                }
                j = _put(quoted, j, 0x5c);
                if (c >= 0x20) {
                    j = _put(quoted, j, c);
                    continue;
                }
                uint256 letter = uint8(_shortEscape(c));
                if (letter != 0) {
                    j = _put(quoted, j, letter);
                    continue;
                }
                j = _put(quoted, j, 0x75);
                j = _put(quoted, j, 0x30);
                j = _put(quoted, j, 0x30);
                j = _put(quoted, j, uint8(HEX_DIGITS[c >> 4]));
                j = _put(quoted, j, uint8(HEX_DIGITS[c & 0x0f]));
            }
            _put(quoted, j, 0x22);
        }
    }

    /// @dev The byte at `i`, which the caller keeps below the length: no bounds check.
    function _byteAt(bytes memory raw, uint256 i) private pure returns (uint256 c) {
        assembly ("memory-safe") {
            c := byte(0, mload(add(add(raw, 0x20), i)))
        }
    }

    /// @dev Writes one byte at `at` and returns the next position.
    function _put(bytes memory out, uint256 at, uint256 c) private pure returns (uint256) {
        assembly ("memory-safe") {
            mstore8(add(add(out, 0x20), at), c)
        }
        return at + 1;
    }

    /// @dev The document without its closing brace, so a body can carry on from it. Keys in
    /// sorted order: answerType, chainId, definitions, evidence, head, question, v, window.
    function _members(Question memory q) private pure returns (bytes memory members) {
        if (q.fromBlock > q.toBlock) revert WindowReversed(q.fromBlock, q.toBlock);
        _checkHexValue(bytes(q.question));
        members = bytes.concat('{"answerType":"', _answerType(q.answerType), '","chainId":', _number(q.chainId));
        if (q.definitions.length > 0) members = bytes.concat(members, ',"definitions":', _definitions(q.definitions));
        if (q.panelEvidence) members = bytes.concat(members, ',"evidence":"panel"');
        if (q.head != 0) members = bytes.concat(members, ',"head":', _number(q.head));
        members = bytes.concat(
            members,
            ',"question":',
            jsonString(q.question),
            ',"v":1,"window":{"fromBlock":',
            _number(q.fromBlock),
            ',"toBlock":',
            _number(q.toBlock),
            "}"
        );
    }

    function _definitions(Definition[] memory definitions) private pure returns (bytes memory object) {
        object = "{";
        for (uint256 i; i < definitions.length; ++i) {
            if (i > 0 && !_sortsBefore(bytes(definitions[i - 1].key), bytes(definitions[i].key))) {
                revert DefinitionsOutOfOrder(i);
            }
            _checkHexValue(bytes(definitions[i].value));
            object = bytes.concat(
                object,
                i == 0 ? bytes("") : bytes(","),
                jsonString(definitions[i].key),
                ":",
                jsonString(definitions[i].value)
            );
        }
        object = bytes.concat(object, "}");
    }

    function _answerType(uint8 answerType) private pure returns (bytes memory) {
        if (answerType == OracleAttestation.ANSWER_BOOL) return "bool";
        if (answerType == OracleAttestation.ANSWER_ADDRESS) return "address";
        if (answerType == OracleAttestation.ANSWER_BYTES32) return "bytes32";
        if (answerType == OracleAttestation.ANSWER_UINT256) return "uint256";
        if (answerType == OracleAttestation.ANSWER_ADDRESS_LIST) return "address[]";
        if (answerType == OracleAttestation.ANSWER_BYTES32_LIST) return "bytes32[]";
        revert UnknownAnswerType(answerType);
    }

    function _number(uint256 value) private pure returns (bytes memory) {
        if (value > MAX_SAFE_INTEGER) revert NumberTooLarge(value);
        return bytes(Strings.toString(value));
    }

    /// @dev The intake's `lowercaseHexValues`: a value that is exactly `0x` and 40 or 64 hex digits is
    /// read in lowercase. One with an uppercase digit is refused here, where it would otherwise hash
    /// as written and never match.
    function _checkHexValue(bytes memory value) private pure {
        uint256 length = value.length;
        if ((length != 42 && length != 66) || value[0] != "0" || value[1] != "x") return;
        bool upper = false;
        for (uint256 i = 2; i < length; ++i) {
            uint8 c = uint8(value[i]);
            if (c >= 0x41 && c <= 0x46) upper = true;
            else if (!((c >= 0x30 && c <= 0x39) || (c >= 0x61 && c <= 0x66))) return;
        }
        if (upper) revert NotLowercaseHex();
    }

    /// @dev `JSON.stringify`'s two-character escapes: backspace, tab, newline, form feed, return.
    function _shortEscape(uint256 c) private pure returns (bytes1) {
        if (c == 0x08) return "b";
        if (c == 0x09) return "t";
        if (c == 0x0a) return "n";
        if (c == 0x0c) return "f";
        if (c == 0x0d) return "r";
        return 0;
    }

    /// @dev Strictly before, byte by byte, a prefix first.
    function _sortsBefore(bytes memory a, bytes memory b) private pure returns (bool) {
        uint256 shorter = a.length < b.length ? a.length : b.length;
        for (uint256 i; i < shorter; ++i) {
            if (a[i] != b[i]) return a[i] < b[i];
        }
        return a.length < b.length;
    }
}
```

## `OracleIntakeConsumer.sol`

Asks on the same chain and is called back by the `Intake`. It imports the protocol's `Intake` contract for its `Callback` struct and `request`; in your project use the `IIntake` interface from the skill instead.

```solidity
// SPDX-License-Identifier: MIT
pragma solidity 0.8.26;

import {Ownable} from "@openzeppelin/contracts/access/Ownable.sol";
import {IERC20} from "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import {SafeERC20} from "@openzeppelin/contracts/token/ERC20/utils/SafeERC20.sol";
import {Intake} from "../Intake.sol";
import {OracleAttestation, OracleAttestationConsumer} from "../OracleAttestation.sol";

/// @title A contract that buys an oracle answer through the Intake and acts on it
/// @notice The worked example behind `docs/onchain-intake.md`. It stores one request body, asks
/// with `ask()`, and receives the signed attestation in `onOracleResult`, which is the callback
/// shape for `oracle.request`: the intake's request id, the attestation struct and the attester's
/// signature over it in this contract's EIP-712 domain. On a version 2 intake, with `failureHook`
/// on, it asks with `requestWithFailure`, so a refused ask or a panel that did not agree calls
/// `onOracleFailure` instead, and the request is no longer pending: the contract knows the price is
/// spent and can ask again. A version 1 intake has no `requestWithFailure`, so the hook is off until
/// the owner moves the contract to version 2 and turns it on.
///
/// It pays in the chain's IMD (an ERC-20 it holds, approved to the intake for exactly one price
/// per ask) or, with `asset` zero, in the native coin it holds.
///
/// Everything that names the protocol is owner-settable and nothing is a constant: the intake
/// address, the action id, the asset and price this contract is willing to pay, and the body.
/// That is the upgrade path. When the plane publishes `oracle-2`, the owner sets the new action id
/// and a body in the new shape; when the price moves, the owner moves `price`; nothing is
/// redeployed.
contract OracleIntakeConsumer is OracleAttestationConsumer, Ownable {
    using SafeERC20 for IERC20;

    error NotTheIntake();
    error UnknownRequest(bytes32 requestId);
    error PanelTooWeak(uint16 panelSize, uint16 agreed);

    event Asked(bytes32 indexed requestId);
    event Answered(bytes32 indexed requestId, bytes answer, uint64 toBlock);
    event Failed(bytes32 indexed requestId, uint8 status, bytes32 reason, uint16 agreed, uint16 answered);

    Intake public intake;
    bytes32 public action;
    /// @notice The asset it pays in: the chain's IMD, or the zero address for the native coin.
    address public asset;
    uint256 public price;
    bytes public body;
    mapping(bytes32 requestId => bool) public pending;
    /// @notice The last answer, `abi.encode`d under the request's answer type.
    bytes public lastAnswer;
    bytes32 public lastQuestionHash;
    /// @notice Why the last request that ended without an answer did: `disagreed`, `body_invalid`, …
    bytes32 public lastFailure;
    /// @notice Whether `ask` names `onOracleFailure`. Only a version 2 intake takes it.
    bool public failureHook;
    /// @notice The smallest panel, and the fewest members agreeing, an answer must come from.
    uint16 public minPanelSize;
    uint16 public minAgreed;

    constructor(
        address owner_,
        Intake intake_,
        bytes32 action_,
        address asset_,
        uint256 price_,
        address oracleSigner_,
        bytes memory body_
    )
        OracleAttestationConsumer(oracleSigner_)
        Ownable(owner_)
    {
        intake = intake_;
        action = action_;
        asset = asset_;
        price = price_;
        body = body_;
    }

    /// @notice Buys one answer. Anyone may trigger it; the contract pays from what it holds.
    function ask() external returns (bytes32 requestId) {
        // The intake is the owner's choice and the price is the owner's ceiling; a caller of ask()
        // chooses neither, which is what makes it safe for anyone to trigger.
        // An ERC-20 is approved for exactly this ask, and the intake pulls it in the same call.
        uint256 value = asset == address(0) ? price : 0;
        if (value == 0) IERC20(asset).forceApprove(address(intake), price);
        Intake.Callback memory callback = Intake.Callback(address(this), this.onOracleResult.selector);
        if (failureHook) {
            // slither-disable-next-line arbitrary-send-eth
            requestId = intake.requestWithFailure{value: value}(action, body, callback, this.onOracleFailure.selector, asset, price);
        } else {
            // slither-disable-next-line arbitrary-send-eth
            requestId = intake.request{value: value}(action, body, callback, asset, price);
        }
        pending[requestId] = true;
        emit Asked(requestId);
    }

    /// @notice The callback for `oracle.request`. Only the intake calls it, only for a request
    /// this contract made, only with an attestation the oracle signer signed for this contract.
    function onOracleResult(bytes32 requestId, OracleAttestation.Attestation calldata a, bytes calldata signature) external {
        if (msg.sender != address(intake)) revert NotTheIntake();
        if (!pending[requestId]) revert UnknownRequest(requestId);
        // Signed with the answer: a panel smaller, or an agreement thinner, than the owner accepts
        // is refused, whatever the body asked for.
        if (a.panelSize < minPanelSize || a.agreed < minAgreed || a.agreed < a.quorum) revert PanelTooWeak(a.panelSize, a.agreed);
        delete pending[requestId];
        _verifyAttestation(a, signature);
        _consume(a.requestId);
        lastAnswer = a.answer;
        lastQuestionHash = a.questionHash;
        emit Answered(requestId, a.answer, a.toBlock);
    }

    /// @notice The failure function: the ask was refused (status 1) or ended without an answer
    /// (status 2). `reason` is a short code, `agreed` and `answered` the panel's counts where there
    /// was one. Nothing here is signed yet, so it only clears the request; it never pays anyone.
    function onOracleFailure(bytes32 requestId, uint8 status, bytes32 reason, uint16 agreed, uint16 answered, bytes calldata)
        external
    {
        if (msg.sender != address(intake)) revert NotTheIntake();
        if (!pending[requestId]) revert UnknownRequest(requestId);
        delete pending[requestId];
        lastFailure = reason;
        emit Failed(requestId, status, reason, agreed, answered);
    }

    // ---- the upgrade path: everything protocol-facing moves without a redeploy ----

    function setIntake(Intake to) external onlyOwner {
        intake = to;
    }

    function setPanelMinimums(uint16 panelSize, uint16 agreed) external onlyOwner {
        minPanelSize = panelSize;
        minAgreed = agreed;
    }

    /// @notice On only against a version 2 intake: a version 1 intake has no `requestWithFailure`.
    function setFailureHook(bool on) external onlyOwner {
        failureHook = on;
    }

    function setAction(bytes32 to, bytes calldata newBody) external onlyOwner {
        action = to;
        body = newBody;
    }

    /// @notice The asset and price move together: a price means nothing in another asset.
    function setPayment(address asset_, uint256 price_) external onlyOwner {
        asset = asset_;
        price = price_;
    }

    /// @notice Takes back what the contract holds to pay with, in `token` or, as zero, the native coin.
    function withdraw(address token, address to, uint256 amount) external onlyOwner {
        if (token == address(0)) {
            (bool ok,) = to.call{value: amount}("");
            require(ok, "withdraw failed");
        } else {
            IERC20(token).safeTransfer(to, amount);
        }
    }

    function setBody(bytes calldata to) external onlyOwner {
        body = to;
    }

    function setSigner(address to) external onlyOwner {
        _setOracleSigner(to);
    }

    receive() external payable {}
}
```

## `OracleBoundConsumer.sol`

Asks one question per request, with the request's subject in it, stores the question's hash and takes only the answer signed for that hash. It imports the protocol's `Intake` for its `Callback` struct and `requestWithFailure`; in your project use the `IIntake` interface from the skill, with `requestWithFailure` added.

```solidity
// SPDX-License-Identifier: MIT
pragma solidity 0.8.26;

import {Ownable} from "@openzeppelin/contracts/access/Ownable.sol";
import {IERC20} from "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import {SafeERC20} from "@openzeppelin/contracts/token/ERC20/utils/SafeERC20.sol";
import {Strings} from "@openzeppelin/contracts/utils/Strings.sol";
import {Intake} from "../Intake.sol";
import {OracleAttestation, OracleAttestationConsumer} from "../OracleAttestation.sol";
import {OracleQuestion} from "../OracleQuestion.sol";

/// @dev Arbitrum's system precompile, at `address(100)` on every Arbitrum chain.
interface IArbSys {
    function arbBlockNumber() external view returns (uint256);
}

/// @title A contract that asks one question per request and acts only on the answer to it
/// @notice The worked example for binding an answer to the request that asked it, in
/// `docs/onchain-intake.md`. `ask(subject)` writes a question of its own with the subject in it,
/// asks it through the `Intake`, and stores the question's hash beside the request. The callback
/// takes an answer only when the oracle signed that same hash. The signature then says which
/// subject the answer is about: an answer to another request's question, delivered to this one by
/// mistake or on purpose, is refused, and so is any answer to a question this contract did not ask.
///
/// The question's window is one block, `WINDOW_DEPTH` behind this chain's head, so its hash is
/// known when it is asked: the plane takes an absolute window once its last block is five deep. On
/// an Arbitrum chain, which Robinhood Chain is, `block.number` is an Ethereum block number, so the
/// owner deploys with `arbitrumBlocks` and the chain's own number comes from `ArbSys`.
///
/// Whoever asks pays: one price in `asset`, approved to this contract, pulled and forwarded in the
/// same call. Nothing is held, so a refused ask spends only the asker's price, and a request that
/// never hears back holds nothing and needs no deadline.
///
/// The wording, what a pass means, the panel and everything protocol-facing are owner-settable. A
/// change applies from the next ask; one already pending keeps the hash it was asked with.
contract OracleBoundConsumer is OracleAttestationConsumer, Ownable {
    using SafeERC20 for IERC20;

    error NotTheIntake();
    error UnknownRequest(bytes32 requestId);
    error WrongQuestion(bytes32 asked, bytes32 answered);
    error PanelTooWeak(uint16 panelSize, uint16 agreed);

    event Asked(bytes32 indexed requestId, bytes32 indexed subject, bytes32 questionHash);
    event Answered(bytes32 indexed requestId, bytes32 indexed subject, bool yes);
    event Failed(bytes32 indexed requestId, bytes32 indexed subject, uint8 status, bytes32 reason);
    event IntakeSet(Intake indexed intake);

    /// @notice How the question reads and who answers it.
    struct Config {
        /// @dev The question is `questionPrefix`, the subject as `0x` and 64 lowercase hex digits,
        /// then `questionSuffix`. ASCII only (`OracleQuestion`).
        string questionPrefix;
        string questionSuffix;
        /// @dev The `pass` definition the panel answers by; empty for none.
        string passMeans;
        uint16 panelSize;
        uint16 quorum;
        uint32 validForSeconds;
    }

    struct Pending {
        bytes32 subject;
        bytes32 questionHash;
    }

    IArbSys internal constant ARB_SYS = IArbSys(address(100));
    /// @notice How many blocks behind the head the window sits: five deep when the plane reads the
    /// ask, with room for the blocks between this transaction and that read.
    uint256 public constant WINDOW_DEPTH = 8;

    /// @notice Whether this is an Arbitrum chain, whose own block number `ArbSys` gives.
    bool public immutable arbitrumBlocks;

    Intake public intake;
    bytes32 public action;
    IERC20 public asset;
    uint256 public price;
    Config internal config;

    mapping(bytes32 requestId => Pending) public pending;
    /// @notice Subjects the oracle answered yes for.
    mapping(bytes32 subject => bool) public passed;

    constructor(
        address owner_,
        Intake intake_,
        bytes32 action_,
        IERC20 asset_,
        uint256 price_,
        address oracleSigner_,
        bool arbitrumBlocks_,
        Config memory config_
    )
        OracleAttestationConsumer(oracleSigner_)
        Ownable(owner_)
    {
        intake = intake_;
        action = action_;
        asset = asset_;
        price = price_;
        arbitrumBlocks = arbitrumBlocks_;
        config = config_;
    }

    /// @notice Asks whether `subject` passed. The caller pays one price.
    function ask(bytes32 subject) external returns (bytes32 requestId) {
        OracleQuestion.Question memory q = question(subject, head() - WINDOW_DEPTH);
        bytes32 questionHash = OracleQuestion.hash(q);
        bytes memory body = OracleQuestion.body(q, config.panelSize, config.quorum, config.validForSeconds);
        asset.safeTransferFrom(msg.sender, address(this), price);
        asset.forceApprove(address(intake), price);
        requestId = intake.requestWithFailure(
            action,
            body,
            Intake.Callback(address(this), this.onOracleResult.selector),
            this.onOracleFailure.selector,
            address(asset),
            price
        );
        pending[requestId] = Pending(subject, questionHash);
        emit Asked(requestId, subject, questionHash);
    }

    /// @notice The callback for `oracle.request`: only from the intake, only for a request this
    /// contract made, and only with the answer to the question that request asked.
    function onOracleResult(bytes32 requestId, OracleAttestation.Attestation calldata a, bytes calldata signature) external {
        if (msg.sender != address(intake)) revert NotTheIntake();
        Pending memory p = pending[requestId];
        // An unknown request reads as zero, and no question hashes to zero. (Slither sees a block
        // number in the hash and calls the comparison dangerous; it is a presence check.)
        // slither-disable-next-line incorrect-equality
        if (p.questionHash == bytes32(0)) revert UnknownRequest(requestId);
        // The binding: the hash is signed with the answer, so this is the answer to this request's
        // question or it is refused, whoever paired the two.
        if (a.questionHash != p.questionHash) revert WrongQuestion(p.questionHash, a.questionHash);
        if (a.panelSize < config.panelSize || a.quorum < config.quorum || a.agreed < a.quorum) {
            revert PanelTooWeak(a.panelSize, a.agreed);
        }
        delete pending[requestId];
        _verifyAttestation(a, signature);
        _consume(a.requestId);
        bool yes = decodeBool(a);
        if (yes) passed[p.subject] = true;
        emit Answered(requestId, p.subject, yes);
    }

    /// @notice The failure function: the ask was refused or ended without an answer. Unsigned, so it
    /// only clears this contract's own request.
    function onOracleFailure(bytes32 requestId, uint8 status, bytes32 reason, uint16, uint16, bytes calldata) external {
        if (msg.sender != address(intake)) revert NotTheIntake();
        Pending memory p = pending[requestId];
        // An unknown request reads as zero, and no question hashes to zero. (Slither sees a block
        // number in the hash and calls the comparison dangerous; it is a presence check.)
        // slither-disable-next-line incorrect-equality
        if (p.questionHash == bytes32(0)) revert UnknownRequest(requestId);
        delete pending[requestId];
        emit Failed(requestId, p.subject, status, reason);
    }

    /// @notice The question `ask(subject)` asks with its window closing at `atBlock`.
    function question(bytes32 subject, uint256 atBlock) public view returns (OracleQuestion.Question memory q) {
        q.question = string.concat(config.questionPrefix, Strings.toHexString(uint256(subject), 32), config.questionSuffix);
        q.chainId = block.chainid;
        q.fromBlock = atBlock;
        q.toBlock = atBlock;
        q.answerType = OracleAttestation.ANSWER_BOOL;
        q.panelEvidence = true;
        if (bytes(config.passMeans).length > 0) {
            q.definitions = new OracleQuestion.Definition[](1);
            q.definitions[0] = OracleQuestion.Definition("pass", config.passMeans);
        }
    }

    /// @notice This chain's own block number.
    function head() public view returns (uint256) {
        return arbitrumBlocks ? ARB_SYS.arbBlockNumber() : block.number;
    }

    function getConfig() external view returns (Config memory) {
        return config;
    }

    // ---- the upgrade path: everything protocol-facing moves without a redeploy ----

    function setConfig(Config calldata to) external onlyOwner {
        config = to;
    }

    /// @notice The only address whose callbacks this contract takes.
    function setIntake(Intake to) external onlyOwner {
        intake = to;
        emit IntakeSet(to);
    }

    function setAction(bytes32 to) external onlyOwner {
        action = to;
    }

    /// @notice The asset and price move together: a price means nothing in another asset.
    function setPayment(IERC20 asset_, uint256 price_) external onlyOwner {
        asset = asset_;
        price = price_;
    }

    function setSigner(address to) external onlyOwner {
        _setOracleSigner(to);
    }
}
```

## `OracleDeliveryReceiver.sol`

Receives an answer bought on another chain, called by `IntakeDelivery` on its own chain.

```solidity
// SPDX-License-Identifier: MIT
pragma solidity 0.8.26;

import {Ownable} from "@openzeppelin/contracts/access/Ownable.sol";
import {OracleAttestation, OracleAttestationConsumer} from "../OracleAttestation.sol";

/// @title A contract that receives an oracle answer bought on another chain
/// @notice The worked example for delivery across chains, in `docs/onchain-intake.md`. Someone
/// asks on chain A's `Intake` with a body that names this contract on chain B:
///
///   "deliver": {"chainId": <B>, "target": "<this contract>", "selector": "<onOracleResult's selector>",
///               "onFailure": "<onOracleFailure's selector>"}
///
/// The plane has the answer signed for this contract on B, and its writer hands it over through
/// `IntakeDelivery` on B. A signature for this contract does not mean this contract's owner asked
/// the question: anyone may pay for a question and name this contract as its receiver, and the
/// answer comes signed for it all the same. Nothing on B can see who asked on A, so the question
/// itself is what this contract checks: it takes an answer only to a question the owner pinned in
/// advance by its `questionHash`, once, from a panel at least as large as the owner requires.
///
/// A question can be pinned only when its hash is known before it is asked, which takes an
/// absolute window (`{"fromBlock", "toBlock"}`, already confirmed when asked): see
/// `OracleQuestion.hash`, or `oracleQuestionHash` in the protocol package. A relative window
/// (`{"hours": N}`) resolves when the plane reads the ask, so its hash cannot be pinned.
///
/// The delivery contract, the signer, the pinned questions and the panel minimums are
/// owner-settable, as everything protocol-facing is.
contract OracleDeliveryReceiver is OracleAttestationConsumer, Ownable {
    error NotTheDelivery();
    error UnexpectedQuestion(bytes32 questionHash);
    error PanelTooWeak(uint16 panelSize, uint16 agreed);

    event Answered(bytes32 indexed sourceRequestId, bytes32 indexed oracleRequestId, bytes answer, uint64 toBlock);
    event Failed(bytes32 indexed sourceRequestId, uint8 status, bytes32 reason, uint16 agreed, uint16 answered);
    event QuestionExpected(bytes32 indexed questionHash, bool expected);
    event PanelMinimumsSet(uint16 panelSize, uint16 agreed);

    address public delivery;
    /// @notice Questions this contract takes an answer to, by `questionHash`; each is answered once.
    mapping(bytes32 questionHash => bool) public expectedQuestion;
    /// @notice The smallest panel, and the fewest members agreeing, an answer must come from.
    uint16 public minPanelSize;
    uint16 public minAgreed;
    /// @notice The last answer, `abi.encode`d under the request's answer type.
    bytes public lastAnswer;
    bytes32 public lastQuestionHash;
    bytes32 public lastSourceRequestId;

    constructor(address owner_, address delivery_, address oracleSigner_)
        OracleAttestationConsumer(oracleSigner_)
        Ownable(owner_)
    {
        delivery = delivery_;
    }

    /// @notice The callback for `oracle.request`, delivered from another chain.
    function onOracleResult(bytes32 sourceRequestId, OracleAttestation.Attestation calldata a, bytes calldata signature) external {
        if (msg.sender != delivery) revert NotTheDelivery();
        // The signature says the oracle answered this question for this contract; only the pin says
        // this contract wanted it answered.
        if (!expectedQuestion[a.questionHash]) revert UnexpectedQuestion(a.questionHash);
        // Signed with the answer: a panel smaller, or an agreement thinner, than the owner accepts
        // is refused, whatever the body asked for.
        if (a.panelSize < minPanelSize || a.agreed < minAgreed || a.agreed < a.quorum) revert PanelTooWeak(a.panelSize, a.agreed);
        _verifyAttestation(a, signature);
        _consume(a.requestId);
        delete expectedQuestion[a.questionHash];
        lastAnswer = a.answer;
        lastQuestionHash = a.questionHash;
        lastSourceRequestId = sourceRequestId;
        emit Answered(sourceRequestId, a.requestId, a.answer, a.toBlock);
    }

    /// @notice The failure function, delivered from another chain when an ask naming this contract
    /// was refused or its panel did not agree. Anyone can cause one, by asking with a bad body and
    /// naming this contract, and nothing in it is signed, so it changes nothing here: it is a log.
    function onOracleFailure(bytes32 sourceRequestId, uint8 status, bytes32 reason, uint16 agreed, uint16 answered, bytes calldata)
        external
    {
        if (msg.sender != delivery) revert NotTheDelivery();
        emit Failed(sourceRequestId, status, reason, agreed, answered);
    }

    /// @notice Pins (or unpins) a question this contract will take an answer to.
    function expectQuestion(bytes32 questionHash, bool expected) external onlyOwner {
        expectedQuestion[questionHash] = expected;
        emit QuestionExpected(questionHash, expected);
    }

    function setPanelMinimums(uint16 panelSize, uint16 agreed) external onlyOwner {
        minPanelSize = panelSize;
        minAgreed = agreed;
        emit PanelMinimumsSet(panelSize, agreed);
    }

    function setDelivery(address to) external onlyOwner {
        delivery = to;
    }

    function setSigner(address to) external onlyOwner {
        _setOracleSigner(to);
    }
}
```

## `OracleConsumerConformance.t.sol`

The conformance test. Copy it into `test/`, replace `deployConsumer` with your contract (deployed at `VECTOR_CONSUMER`, oracle signer `SIGNER`, trusting `caller()`), point `caller()` at whatever calls your callback (a mock intake for a same-chain consumer), and keep every vector constant as it is. Rename `onOracleResult` in `CALLBACK` if your callback has another name. If your consumer cannot use `IntakeDelivery`, drop that import and the delivered test, and keep the other three.

```solidity
// SPDX-License-Identifier: MIT
pragma solidity 0.8.26;

import {Test} from "forge-std/Test.sol";
import {OracleAttestation} from "../src/OracleAttestation.sol";
import {IntakeDelivery} from "../src/IntakeDelivery.sol";

/// @notice Delivered whole to builders by the `oracle-consumer` skill. It is the one test that ties
/// a consumer to what the oracle actually signs: the protocol's vector values, its digest and a
/// signature made by `oracle-eip712.ts` (the same vector `OracleAttestation.t.sol` pins). A consumer
/// whose struct, type string or domain differs from the protocol's passes its own self-signed tests
/// and fails this one.

/// `SIGNER` as its oracle signer and `caller()` as the address its callback trusts, and `CALLBACK`

    string constant CALLBACK =

        vm.warp(ISSUED_AT);
        delivery = new IntakeDelivery(address(this), WRITER);

    function deployConsumer() internal {

        consumer.expectQuestion(vector().questionHash, true);

    /// `Intake` (or a mock of it) for one from the same chain.
    function caller() internal view returns (address) {
        return address(delivery);
    }

    /// @dev The vector attestation: a `bytes32[]` answer with one element, figure 12345, blocks
    /// 100–200, a panel of five with a quorum of four that agreed unanimously.
    function vector() internal pure returns (OracleAttestation.Attestation memory a) {
        bytes32[] memory ids = new bytes32[](1);
        ids[0] = bytes32(uint256(1));
        a = OracleAttestation.Attestation({
            requestId: 0x0000000000004000800000000000000100000000000000000000000000000000,
            chainId: 1,
            questionHash: 0x2117f4362ebfa37aa8a8c0fed548604fe09ac46faf8ae7559cd64780f26a46fb,
            answerType: OracleAttestation.ANSWER_BYTES32_LIST,
            answer: abi.encode(ids),
            figure: 12345,
            fromBlock: 100,
            toBlock: 200,
            blockHash: bytes32(uint256(7)),
            panelJobId: 0x0000000000004000800000000000000200000000000000000000000000000000,
            panelSize: 5,
            quorum: 4,
            agreed: 5,
            issuedAt: ISSUED_AT,
            expiresAt: EXPIRES_AT
        });
    }

    /// @notice The digest your consumer verifies is the one the oracle signs.
    function test_digestMatchesTheProtocol() public view {
        assertEq(consumer.attestationDigest(vector()), VECTOR_DIGEST, "struct, type string or domain differs from the protocol's");
    }

    /// @notice Your callback's selector is the one the intake calls.
    function test_callbackSelectorIsCanonical() public view {
        assertEq(consumer.onOracleResult.selector, bytes4(keccak256(bytes(CALLBACK))), "callback parameters differ from the protocol's");
    }

    /// @notice The protocol's own signature, delivered the way the writer delivers it, is accepted.
    /// @dev A consumer that refuses the vector's answer type signs its own answer with `SIGNER_KEY`

    function test_acceptsTheProtocolSignatureAsDelivered() public {
        bytes32 sourceRequestId = keccak256("source request");

            SOURCE_INTAKE,
            sourceRequestId,
            VECTOR_CONSUMER,

        assertTrue(consumer.consumed(vector().requestId));

    function test_acceptsAFreshSignatureFromTheVectorKey() public {
        OracleAttestation.Attestation memory a = vector();
        a.requestId = keccak256("another request");
        (uint8 v, bytes32 r, bytes32 s) = vm.sign(SIGNER_KEY, consumer.attestationDigest(a));
        vm.prank(caller());
        consumer.onOracleResult(bytes32(0), a, abi.encodePacked(r, s, v));
        assertTrue(consumer.consumed(a.requestId));
    }
}
```
